basics
    all multi-byte integers: little endian
    all signed integers: two's complement
    all floating point values: 8 bytes, IEEE-754 binary64
    all explicit padding bytes: 0x00
    all explicit reserved bytes: 0x00
    page size: 4096 bytes
    invalid page id: 4 bytes, 0xFF 0xFF 0xFF 0xFF
    invalid slot id: 2 bytes, 0xFF 0xFF
    first allocatable page id: 4 bytes, value 1

logical type code bytes
    BOOLEAN: 1 byte, value 0x01
    BYTE: 1 byte, value 0x02
    INT32: 1 byte, value 0x03
    INT64: 1 byte, value 0x04
    DOUBLE: 1 byte, value 0x05
    STRING: 1 byte, value 0x06

logical value bytes
    BOOLEAN:
        1 byte: 0x00 false, 0x01 true
    BYTE:
        1 byte: signed integer
    INT32:
        4 bytes: signed integer
    INT64:
        8 bytes: signed integer
    DOUBLE:
        8 bytes: floating point value
    STRING(N):
        N bytes: UTF-8 bytes followed by 0x00 padding

metadata text
    2 bytes: byte length
    byte length bytes: UTF-8 bytes
    padding bytes: align next offset to 4

file tree
    catalog.meta
    databases/
        for each database:
            <database name>/
                database.meta
                tables.meta
                indexes.meta
                data.pages
                index.pages

cross-file reads
    row values from data.pages:
        stored in data.pages: row bytes
        needed from tables.meta: column count, logical type codes, string capacities, column order

    table rows for one table:
        stored in tables.meta: first table page id, last table page id
        stored in data.pages: next table page id, slot directory, row bytes

    table indexes:
        stored in tables.meta: no index list
        needed from indexes.meta: scan index entries by table id

    index key values from index.pages:
        stored in index.pages: key bytes
        needed from indexes.meta: logical type code, string capacity, key size

    B+Tree root:
        stored in indexes.meta: root page id
        stored in index.pages: leaf pages and internal pages

    index leaf RID target row:
        stored in index.pages: RID bytes
        needed from data.pages: table page slot bytes
        needed from tables.meta: row schema

    deleted row original payload:
        stored in data.pages slot: row size 0
        original row size after delete: not stored
        original row decode after delete: not available from files

    current selected database:
        not stored in files

    buffer pool state:
        not stored in files

    SQL parser, binder, planner, execution state:
        not stored in files

catalog.meta
    offset 0, 4 bytes: ASCII "DCAT" (magic)
    offset 4, 4 bytes: database count
    offset 8:
        for each database:
            2 bytes: database name byte length
            database name byte length bytes: database name UTF-8 bytes
            padding bytes: align next database entry to 4

databases/<database name>/database.meta
    offset 0, 4 bytes: ASCII "DDBD" (magic)
    offset 4:
        2 bytes: database name byte length
        database name byte length bytes: database name UTF-8 bytes
        padding bytes: align next offset to 4
    next offset, 4 bytes: reserved

databases/<database name>/tables.meta
    offset 0, 4 bytes: ASCII "DTBL" (magic)
    offset 4, 4 bytes: next table id
    offset 8, 4 bytes: table count
    offset 12, 4 bytes: reserved
    offset 16:
        for each table:
            offset + 0, 4 bytes: table id
            offset + 4, 4 bytes: first table page id
            offset + 8, 4 bytes: last table page id
            offset + 12, 1 byte: column count
            offset + 13, 3 bytes: padding
            offset + 16:
                2 bytes: table name byte length
                table name byte length bytes: table name UTF-8 bytes
                padding bytes: align next offset to 4
            next offset:
                for each column:
                    offset + 0, 1 byte: logical type code
                    offset + 1, 1 byte: column flags
                    offset + 2, 2 bytes: string capacity in bytes
                    offset + 4:
                        2 bytes: column name byte length
                        column name byte length bytes: column name UTF-8 bytes
                        padding bytes: align next column entry to 4
            next offset:
                padding bytes: align next table entry to 8

    column flags byte
        bit 0: nullable
        bit 1: primary key
        bits 2..7: unused

row payload inside data.pages table slots
    row start offset: slot row offset
    row size: slot row size

    relative offset 0, 1 byte: column count
    relative offset 1, B bytes: null bitmap
        B = (column count + 7) / 8
    next relative offset:
        padding bytes: align next relative offset to 8
    next relative offset:
        for each column in table schema order:
            padding bytes: align current relative offset to column alignment
            value bytes: logical value bytes for that column type
    next relative offset:
        padding bytes: align row size to 8

    null bitmap byte layout
        byte index: column index / 8
        bit index: column index % 8
        bit value 0: value bytes contain a non-null value
        bit value 1: value bytes are 0x00 bytes
        unused bits in final byte: 0

    column value alignment
        BOOLEAN: 1
        BYTE: 1
        INT32: 4
        INT64: 8
        DOUBLE: 8
        STRING(N): 1

RID stored in index leaf entries
    offset + 0, 4 bytes: data.pages page id
    offset + 4, 2 bytes: table page slot id
    offset + 6, 2 bytes: padding

databases/<database name>/data.pages
    page offset:
        page id * 4096

    page 0
        offset 0, 4 bytes: ASCII "DPAG"
        offset 4, 4 bytes: page size, value 4096
        offset 8, 4 bytes: page count
        offset 12, 4 bytes: first free page id
        offset 16, 4 bytes: free page count
        offset 20, 4076 bytes: reserved

    free page
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0xFF 0xFF
        offset 6, 2 bytes: padding
        offset 8, 4 bytes: next free page id
        offset 12, 4084 bytes: unused

    table page
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0x01 0x00
        offset 6, 2 bytes: padding
        offset 8, 4 bytes: next table page id
        offset 12, 2 bytes: slot count
        offset 14, 2 bytes: free start offset
        offset 16, 2 bytes: free end offset
        offset 18, 2 bytes: padding
        offset 20, 4 bytes: reserved
        offset 24:
            for each slot:
                offset + 0, 2 bytes: row offset
                offset + 2, 2 bytes: row size
        free start offset:
            first byte after slot directory
        free end offset:
            first byte of row payload area
        row payload area:
            bytes free end offset through 4095

    empty table page bytes
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0x01 0x00
        offset 6, 2 bytes: padding
        offset 8, 4 bytes: invalid page id
        offset 12, 2 bytes: value 0
        offset 14, 2 bytes: value 24
        offset 16, 2 bytes: value 4096
        offset 18, 2 bytes: padding
        offset 20, 4 bytes: reserved
        offset 24, 4072 bytes: 0x00

    deleted row slot
        offset + 0, 2 bytes: previous row offset
        offset + 2, 2 bytes: value 0

databases/<database name>/indexes.meta
    offset 0, 4 bytes: ASCII "DIXM"
    offset 4, 4 bytes: next index id
    offset 8, 4 bytes: index count
    offset 12, 4 bytes: reserved
    offset 16:
        for each index:
            offset + 0, 4 bytes: index id
            offset + 4, 4 bytes: table id
            offset + 8, 4 bytes: root page id in index.pages
            offset + 12, 1 byte: key column count
            offset + 13, 1 byte: column index
            offset + 14, 1 byte: logical type code
            offset + 15, 1 byte: index flags
            offset + 16, 2 bytes: string capacity in bytes
            offset + 18, 2 bytes: padding
            offset + 20:
                2 bytes: index name byte length
                index name byte length bytes: index name UTF-8 bytes
                padding bytes: align next offset to 4
            next offset:
                2 bytes: column name byte length
                column name byte length bytes: column name UTF-8 bytes
                padding bytes: align next index entry to 8

    index flags byte
        bit 0: unique
        bit 1: primary key backing index
        bits 2..7: unused

index key bytes
    relative offset 0, 1 byte: null bitmap
    next relative offset:
        padding bytes: align next relative offset to indexed value alignment
    next relative offset:
        value bytes: logical value bytes for indexed column type
    next relative offset:
        padding bytes: align key size to 4

    null bitmap byte
        bit 0 value 0: key contains a non-null value
        bit 0 value 1: value bytes are 0x00 bytes
        bits 1..7: 0

    indexed value alignment
        BOOLEAN: 1
        BYTE: 1
        INT32: 4
        INT64: 8
        DOUBLE: 8
        STRING(N): 1

databases/<database name>/index.pages
    page offset:
        page id * 4096

    page 0
        offset 0, 4 bytes: ASCII "IPAG"
        offset 4, 4 bytes: page size, value 4096
        offset 8, 4 bytes: page count
        offset 12, 4 bytes: first free page id
        offset 16, 4 bytes: free page count
        offset 20, 4076 bytes: reserved

    free page
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0xFF 0xFF
        offset 6, 2 bytes: padding
        offset 8, 4 bytes: next free page id
        offset 12, 4084 bytes: unused

    common index page header
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind
        offset 6, 2 bytes: padding
        offset 8, 2 bytes: key count
        offset 10, 2 bytes: key size
        offset 12, 4 bytes: parent page id
        offset 16, 4 bytes: next leaf page id
        offset 20, 4 bytes: first child page id
        offset 24, 4 bytes: reserved
        offset 28, 4 bytes: reserved

    leaf page
        offset 0, 32 bytes: common index page header
            page kind: 0x02 0x00
            next leaf page id: next leaf page id or invalid page id
            first child page id: invalid page id
        offset 32:
            for each leaf entry:
                offset + 0, key size bytes: index key bytes
                offset + key size, 4 bytes: data.pages page id
                offset + key size + 4, 2 bytes: table page slot id
                offset + key size + 6, 2 bytes: padding

    leaf entry size
        key size + 8 bytes

    leaf max entry count
        floor((4096 - 32) / leaf entry size)

    empty leaf page bytes
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0x02 0x00
        offset 6, 2 bytes: padding
        offset 8, 2 bytes: value 0
        offset 10, 2 bytes: key size
        offset 12, 4 bytes: invalid page id
        offset 16, 4 bytes: invalid page id
        offset 20, 4 bytes: invalid page id
        offset 24, 8 bytes: reserved
        offset 32, 4064 bytes: 0x00

    internal page
        offset 0, 32 bytes: common index page header
            page kind: 0x03 0x00
            next leaf page id: invalid page id
            first child page id: leftmost child page id
        offset 32:
            for each internal entry:
                offset + 0, key size bytes: separator index key bytes
                offset + key size, 4 bytes: right child page id

    internal entry size
        key size + 4 bytes

    internal max entry count
        floor((4096 - 32) / internal entry size)

    empty internal page bytes
        offset 0, 4 bytes: page id
        offset 4, 2 bytes: page kind, value 0x03 0x00
        offset 6, 2 bytes: padding
        offset 8, 2 bytes: value 0
        offset 10, 2 bytes: key size
        offset 12, 4 bytes: invalid page id
        offset 16, 4 bytes: invalid page id
        offset 20, 4 bytes: invalid page id
        offset 24, 8 bytes: reserved
        offset 32, 4064 bytes: 0x00
